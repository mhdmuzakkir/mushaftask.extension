"إِنَّمَا هَذِهِ الْقُلُوبُ أَوْعِيَةٌ، فَاشْغَلُوهَا بِالْقُرْآنِ وَلَا تَشْغَلُوهَا بِغَيْرِهِ":"عَبْدُ اللَّهِ بْنُ مَسْعُودٍ -رضي الله عنه-":"مصنف ابن أبي شيبة (7/103)، إسناد صحيح"

"إِذَا أَرَدْتُمُ الْعِلْمَ فَانْثُرُوا الْقُرْآنَ، فَإِنَّ فِيهِ عِلْمَ الْأَوَّلِينَ وَالْآخِرِينَ":"عَبْدُ اللَّهِ بْنُ مَسْعُودٍ -رضي الله عنه-":"مصنف عبد الرزاق (4742)، الزهد لابن أبي عاصم (78)، المعجم الأوسط للطبراني (8755)، الشعب للبيهقي (3/84)، إسناد صحيح"

"تَعَوَّدُوا الْخَيْرَ، فَإِنَّمَا الْخَيْرُ فِي الْعَادَةِ":"عَبْدُ اللَّهِ بْنُ مَسْعُودٍ -رضي الله عنه-":"مصنف ابن أبي شيبة (37291)، صحيح الترغيب للألباني"

"لَا تَهُذُّوا الْقُرْآنَ هُذَّ الشِّعْرِ، وَلَا تَنْثُرُوهُ نَثْرَ الدَّقَلِ، قِفُوا عِنْدَ عَجَائِبِهِ، وَحَرِّكُوا بِهِ الْقُلُوبَ":"عَبْدُ اللَّهِ بْنُ مَسْعُودٍ -رضي الله عنه-":"الشعب للبيهقي"

"إِنَّا لَقَدْ عَشِينَا دَهْرًا طَوِيلًا وَأَحَدُنَا يُؤْتَى الْإِيمَانَ قَبْلَ الْقُرْآنِ":"عَبْدُ اللَّهِ بْنُ مَسْعُودٍ -رضي الله عنه-":"المسند لأحمد (1/379)"

"إِنَّ الْبَيْتَ لَيَتَّسِعُ عَلَى أَهْلِهِ وَتَحْضُرُهُ الْمَلَائِكَةُ وَتَهْجُرُهُ الشَّيَاطِينُ وَيَكْثُرُ خَيْرُهُ أَنْ يُقْرَأَ فِيهِ الْقُرْآنُ":"أَبُو هُرَيْرَةَ -رضي الله عنه-":"مصنف ابن أبي شيبة (7/103)"

"لَوْ طَهُرَتْ قُلُوبُنَا لَمْ تَشْبَعْ مِنْ قِرَاءَةِ كِتَابِ اللَّهِ":"عُثْمَانُ بْنُ عَفَّانَ وَحُذَيْفَةُ بْنُ الْيَمَانِ -رضي الله عنهما-":"الطبر للذهبي، تاريخ بغداد للخطيب البغدادي"

"كُنَّا نَحْفَظُ الْعَشَرَ آيَاتٍ فَلَا نَنْتَقِلُ إِلَى مَا بَعْدَهَا حَتَّى نَعْمَلَ بِهِنَّ":"عُمَرُ بْنُ الْخَطَّابِ -رضي الله عنه-":"المعجم الأوسط للطبراني (8755)، إسناد حسن"

"لَأَنْ أَقْرَأَ الْبَقَرَةَ وَآلَ عِمْرَانَ أُرَتِّلُهُمَا وَأَتَدَبَّرُهُمَا أَحَبُّ إِلَيَّ مِنْ أَنْ أَقْرَأَ الْقُرْآنَ هُذَرْمَةً":"عَبْدُ اللَّهِ بْنُ عَبَّاسٍ -رضي الله عنهما-":"المسند لأحمد (1/244)"

"إِيَّاكُمْ وَالَّذِينَ يُحَرِّفُونَ الْقُرْآنَ، وَإِيَّاكُمْ وَالْهَذَاذِينَ بِالْقُرْآنِ... فَإِنَّمَا مَثَلُ ذَٰلِكَ كَمَثَلِ الْأَكَمَةِ لَا أَمْسَكَتْ مَاءً وَلَا أَنْبَتَتْ كَلَأً":"أَبُو الدَّرْدَاءِ -رضي الله عنه-":"مصنف ابن أبي شيبة (7/103)"

"إِنَّ مِنْ كَانَ قَبْلَكُمْ رَأَوْا الْقُرْآنَ رَسَائِلَ مِنْ رَبِّهِمْ فَكَانُوا يَتَدَبَّرُونَهَا بِاللَّيْلِ وَيَتَفَقَّدُونَهَا فِي النَّهَارِ":"الْحَسَنُ بْنُ عَلِيٍّ -رضي الله عنهما-":"حلية الأولياء لأبي نعيم"

"إِنَّ لِلَّهِ أَهْلِينَ مِنَ النَّاسِ... هُمْ أَهْلُ الْقُرْآنِ، أَهْلُ اللَّهِ وَخَاصَّتُهُ":"النَّبِيُّ ﷺ -رواه أَنَسُ بْنُ مَالِكٍ -رضي الله عنه-":"المسند لأحمد (11870)، سُنَنُ ابْنِ مَاجَهْ (215)، صحيح ابن ماجه للألباني، صحيح الجامع (2165)"

"يَنْبَغِي لِحَامِلِ الْقُرْآنِ أَنْ يُعْرَفَ بِلَيْلِهِ إِذَا النَّاسُ نَائِمُونَ، وَبِنَهَارِهِ إِذَا النَّاسُ يَفْرَطُونَ، وَبِحُزْنِهِ إِذَا النَّاسُ يَفْرَحُونَ، وَبِبُكَائِهِ إِذَا النَّاسُ يَضْحَكُونَ، وَبِصَمْتِهِ إِذَا النَّاسُ يَخُوضُونَ، وَبِخُشُوعِهِ إِذَا النَّاسُ يَخْتَالُونَ":"عَبْدُ اللَّهِ بْنُ مَسْعُودٍ -رضي الله عنه-":"حلية الأولياء لأبي نعيم"

"حَامِلُ الْقُرْآنِ حَامِلُ رَايَةِ الْإِسْلَامِ، لَا يَنْبَغِي أَنْ يَلْهُوَ مَعَ مَنْ يَلْهُوَ، وَلَا يَسْهُوَ مَعَ مَنْ يَسْهُوَ، وَلَا يَلْغُوَ مَعَ مَنْ يَلْغُوَ، تَعْظِيمًا لِحَقِّ الْقُرْآنِ":"الْفُضَيْلُ بْنُ عِيَاضٍ -رحمه الله-":"حلية الأولياء (7/33)"

"يَنْبَغِي لِحَامِلِ الْقُرْآنِ أَنْ لَا يَكُونَ لَهُ إِلَى أَحَدٍ حَاجَةٌ، وَلَا يَكُونُ لَهُ حَاجَةٌ إِلَى الْخُلَفَاءِ وَمَنْ دُونَهُمْ، فَإِنَّهُ يَنْبَغِي أَنْ تَكُونَ حَوَائِجُ الْخَلْقِ إِلَيْهِ":"الْفُضَيْلُ بْنُ عِيَاضٍ -رحمه الله-":"حلية الأولياء لأبي نعيم"

"سَيَبْلَى الْقُرْآنُ فِي صُدُورِ أَقْوَامٍ كَمَا يَبْلَى الثَّوْبُ، فَيَتَهَافَتُ يَقْرَؤُونَهُ لَا يَجِدُونَ لَهُ لَذَّةً...":"مُعَاذُ بْنُ جَبَلٍ -رضي الله عنه-":"مصنف ابن أبي شيبة (7/103)"

"إِنَّا أَنْزَلَ اللَّهُ عَلَيْهِمْ لِيَعْمَلُوا بِهِ، فَاتَّخَذُوا دِرَاسَتَهُ عَمَلًا":"عَبْدُ اللَّهِ بْنُ مَسْعُودٍ -رضي الله عنه-":"مصنف ابن أبي شيبة"

"رُبَّ تَالٍ لِلْقُرْآنِ وَالْقُرْآنُ يَلْعَنُهُ":"أَنَسُ بْنُ مَالِكٍ -رضي الله عنه-":"حلية الأولياء (7/103)"

"خُذُوا مِنَ الْأَعْمَالِ مَا تُطِيقُونَ، فَإِنَّ اللَّهَ لَا يَمَلُّ حَتَّى تَمَلُّوا":"النَّبِيُّ ﷺ -رواه عَائِشَةَ -رضي الله عنها-":"صحيح البخاري (1150)، صحيح مسلم (782)"

"يَا عَبْدَ اللَّهِ، لَا تَكُنْ مِثْلَ فُلَانٍ كَانَ يَقُومُ اللَّيْلَ فَتَرَكَ قِيَامَ اللَّيْلِ":"النَّبِيُّ ﷺ -رواه عَبْدُ اللَّهِ بْنُ عَمْرٍو -رضي الله عنهما-":"صحيح البخاري (1150)، صحيح مسلم (1159)"

"إِنَّمَا الْأَعْمَالُ بِالْخَوَاتِيمِ":"النَّبِيُّ ﷺ":"صحيح البخاري (6606)، صحيح مسلم (2378)"

"كُونُوا يَنَابِيعَ الْعِلْمِ، مَصَابِيحَ الْهُدَى، أَحْلَاسَ الْبُيُوتِ، سُرُجَ اللَّيْلِ، جُدُدَ الْقُلُوبِ... تُعْرَفُونَ فِي السَّمَاءِ وَتَخْفَوْنَ عَلَى أَهْلِ الْأَرْضِ":"عَبْدُ اللَّهِ بْنُ مَسْعُودٍ -رضي الله عنه-":"المسند لأحمد (1/379)، الطبراني، البيهقي"

"مَا نَزَلَتْ آيَةٌ مِنْ كِتَابِ اللَّهِ إِلَّا وَأَنَا أَعْلَمُ أَيْنَ نُزِلَتْ، وَأَعْلَمُ فِيمَ نُزِلَتْ":"عَبْدُ اللَّهِ بْنُ عَبَّاسٍ -رضي الله عنهما-":"صحيح البخاري (ترجمة ابن عباس)"

"تَلَمَّسُوا الْخَيْرَ دَهْرَكُمْ كُلَّهُ، وَتَعَرَّضُوا لِنَفَحَاتِ رَحْمَةِ اللَّهِ":"أَبُو الدَّرْدَاءِ -رضي الله عنه-":"حلية الأولياء (1/255)"

"أَفْضَلُ الْأَعْمَالِ: أَدَاءُ مَا افْتَرَضَ اللَّهُ، وَالْوَرَعُ عَمَّا حَرَّمَ اللَّهُ، وَصِدْقُ النِّيَّةِ فِيمَا عِنْدَ اللَّهِ":"عُمَرُ بْنُ الْخَطَّابِ -رضي الله عنه-":"الورع لابن أبي الدنيا"

"إِنْ أَصْبَحْتَ فَلَا تَنْتَظِرِ الْمَسَاءَ، وَإِنْ أَمْسَيْتَ فَلَا تَنْتَظِرِ الصَّبَاحَ":"عَبْدُ اللَّهِ بْنُ عُمَرَ -رضي الله عنهما-":"صحيح البخاري (6416)"

"إِنِّي لَأُحِبُّ أَنْ أَرَى الرَّجُلَ فِي الشِّتَاءِ يَبْتَسِمُ":"عَبْدُ اللَّهِ بْنُ مَسْعُودٍ -رضي الله عنه-":"الصبر والشكر لابن أبي الدنيا"

"أَفَلَا أَكُونُ عَبْدًا شَكُورًا؟":"النَّبِيُّ ﷺ -رواه عَائِشَةَ -رضي الله عنها-":"صحيح البخاري (1130)، صحيح مسلم (2820)"

"اللَّهُمَّ أَعِنِّي عَلَى ذِكْرِكَ وَشُكْرِكَ وَحُسْنِ عِبَادَتِكَ":"النَّبِيُّ ﷺ -رواه مُعَاذُ بْنُ جَبَلٍ -رضي الله عنه-":"سُنَنُ أَبِي دَاوُد (1511)، السنن الكبرى للنسائي (1298)"

"قَيِّدُوا نِعَمَ اللَّهِ بِشُكْرِ اللَّهِ":"عُمَرُ بْنُ عَبْدِ الْعَزِيزِ -رحمه الله-":"تاريخ ابن عساكر"

"إِنَّ النِّعْمَةَ مَوْصُولَةٌ بِالشُّكْرِ، وَالشُّكْرُ يَتَّصِلُ بِالْمَزِيدِ، وَهُمَا مُقَرُونَانِ فِي قَرْنٍ، فَلَنْ يَنْقَطِعَ الْمَزِيدُ مِنَ اللَّهِ حَتَّى يَنْقَطِعَ الشُّكْرُ مِنَ الْعَبْدِ":"عَلِيُّ بْنُ أَبِي طَالِبٍ -رضي الله عنه-":"البحر الرائق لأحمد فريد، الآداب الشرعية لابن مفلح"

"مَا أَنْعَمَ اللَّهُ عَلَى عَبْدٍ نِعْمَةً فَقَالَ: الْحَمْدُ لِلَّهِ، إِلَّا كَانَ مَا أُعْطِيَ أَكْثَرَ مِمَّا أُخِذَ":"الْحَسَنُ الْبَصْرِيُّ -رحمه الله-":"الشكر لابن أبي الدنيا"

"مَا شُكْرُ الْعَيْنَيْنِ؟ قَالَ: إِنْ رَأَيْتَ بِهِمَا خَيْرًا أَعْلَنْتَهُ، وَإِنْ رَأَيْتَ بِهِمَا شَرًّا سَتَرْتَهُ":"أَبُو حَازِمٍ السَّلْمِيُّ -رحمه الله-":"الزهد لأحمد بن حنبل، حلية الأولياء"

"لَمْ يَزَلْ يَحْمَدُ اللَّهَ حَتَّى إِذَا لَمْ يَبْقَ مِنْ نِعَمِهِ شَيْءٌ إِلَّا عَيْنَاهُ وَلِسَانُهُ وَيَدَاهُ وَرِجْلَاهُ":"سَلْمَانُ الْفَارِسِيُّ -رضي الله عنه-":"الشكر لابن أبي الدنيا، حلية الأولياء"

"مَا أُرِيدُ بِهِ وَجْهِيَ فَقَلِيلُهُ كَثِيرٌ، وَمَا أُرِيدُ بِهِ غَيْرِيَ فَكَثِيرُهُ قَلِيلٌ":"أَبُو هُرَيْرَةَ -رضي الله عنه-":"مصنف ابن أبي شيبة"

"لَقَدْ عَشِينَا دَهْرًا طَوِيلًا وَأَحَدُنَا يُؤْتَى الْإِيمَانَ قَبْلَ الْقُرْآنِ، فَتَنْزِلُ السُّورَةُ عَلَى مُحَمَّدٍ ﷺ فَيَتَعَلَّمُ حَلَالَهَا وَحَرَامَهَا وَآمِرَهَا وَزَاجِرَهَا... ثُمَّ لَقَدْ رَأَيْتُ رِجَالًا يُؤْتَى أَحَدُهُمُ الْقُرْآنُ قَبْلَ الْإِيمَانِ، فَيَقْرَأُ مَا بَيْنَ الْفَاتِحَةِ إِلَى خَاتِمَتِهِ لَا يَدْرِي مَا آمِرُهُ وَلَا زَاجِرُهُ، يَنْثُرُهُ نَثْرَ الدَّقَلِ":"عَبْدُ اللَّهِ بْنُ عَمْرِو بْنِ الْعَاصِ -رضي الله عنهما-":"المسند لأحمد (2/192)"

"إِنَّ الْعَبْدَ يَزْدَادُ فِي الدُّنْيَا نُقْصَانًا، وَيَزْدَادُ فِي الْآخِرَةِ نُقْصَانًا، حَتَّى يَكُونَ شَرَّ أَهْلِ الْأَرْضِ":"عَبْدُ اللَّهِ بْنُ مَسْعُودٍ -رضي الله عنه-":"مصنف ابن أبي شيبة"

"وَاللَّهِ لَا يُؤْمِنُ عَبْدٌ بِهَذَا الْقُرْآنِ إِلَّا حُزِنَ وَذَبَلَ، وَإِلَّا نَصِبَ وَإِلَّا ذَابَ وَإِلَّا تَعِبَ":"الْحَسَنُ الْبَصْرِيُّ -رحمه الله-":"حلية الأولياء (2/158)"

"لَوْ طَهُرَتْ قُلُوبُنَا لَمْ تَشْبَعْ مِنْ قِرَاءَةِ الْقُرْآنِ":"عُثْمَانُ بْنُ عَفَّانَ -رضي الله عنه-":"حلية الأولياء (1/78)"

"كُلَّمَا زَادَ حِزْبِي مِنَ الْقُرْآنِ زَادَتِ الْبَرَكَةُ فِي وَقْتِي":"عَبْدُ اللَّهِ بْنُ مَسْعُودٍ -رضي الله عنه-":"مصنف ابن أبي شيبة"

"أَكْثِرْ مِنْ قِرَاءَةِ الْقُرْآنِ وَلَا تَتْرُكْهُ؛ فَإِنَّهُ يَتَيَسَّرُ لَكَ الَّذِي تَطْلُبُهُ عَلَى قَدْرِ مَا تَقْرَأُ":"إِبْرَاهِيمُ الْمَقْدِسِيُّ -رحمه الله- (لمحمد بن الضياء المقدسي)":"حلية الأولياء (9/63)، سير أعلام النبلاء (20/414)"

"وَاللَّهِ مَا دُونَ الْقُرْآنِ مِنْ غِنًى وَلَا بَعْدَهُ مِنْ فَاقَةٍ":"الْحَسَنُ الْبَصْرِيُّ -رحمه الله-":"مصنف ابن أبي شيبة، حلية الأولياء (2/158)"

"وَإِذَا أَرَدْتَ صَلَاحَ قَلْبِكَ، أَوْ ابْنِكَ، أَوْ أَخِيكَ، أَوْ مَنْ شِئْتَ صَلَاحَهُ، فَأَوْدِعْهُ فِي رِيَاضِ الْقُرْآنِ وَبَيْنَ صُحْبَةِ الْقُرْآنِ، سَيُصْلِحُهُ اللَّهُ شَاءَ أَمْ أَبَى":"الْإِمَامُ الشَّافِعِيُّ -رحمه الله-":"سير أعلام النبلاء للذهبي"

"لَيْسَ حِفْظُ الْقُرْآنِ بِحِفْظِ الْحُرُوفِ، وَلَكِنْ إِقَامَةُ حُدُودِهِ":"عَبْدُ اللَّهِ بْنُ مَسْعُودٍ -رضي الله عنه-":"مصنف ابن أبي شيبة"

"إِنَّ لِلَّهِ أَهْلِينَ مِنَ النَّاسِ... هُمْ أَهْلُ الْقُرْآنِ، أَهْلُ اللَّهِ وَخَاصَّتُهُ":"أَنَسُ بْنُ مَالِكٍ -رضي الله عنه- (رواية الحديث عن النبي ﷺ)":"المسند لأحمد (11870)، سُنَنُ ابْنِ مَاجَهْ (215)، صحيح الجامع (2165)"

"قُرَّاءُ الْقُرْآنِ ثَلَاثَةٌ: فَصِنْفٌ اتَّخَذُوهُ بِضَاعَةً يَأْكُلُونَ بِهَا، وَصِنْفٌ أَقَامُوا حُرُوفَهُ وَضَيَّعُوا حُدُودَهُ... وَصِنْفٌ عَمِدُوا إِلَى دَوَاءِ الْقُرْآنِ فَوَضَعُوهُ عَلَى دَاءِ قُلُوبِهِمْ... فَأُولَٰئِكَ الَّذِينَ يُسَقِي اللَّهُ بِهِمُ الْغَيْثَ، وَيَنْصُرُ بِهِمْ عَلَى الْأَعْدَاءِ":"الْحَسَنُ الْبَصْرِيُّ -رحمه الله-":"حلية الأولياء (2/158)"

"كَانَ يُقَالُ: خَمْسٌ كَانَ عَلَيْهَا أَصْحَابُ مُحَمَّدٍ ﷺ وَالْتَابِعُونَ: لُزُومُ الْجَمَاعَةِ، وَاتِّبَاعُ السُّنَّةِ، وَعِمَارَةُ الْمَسْجِدِ، وَتِلَاوَةُ الْقُرْآنِ، وَالْجِهَادُ فِي سَبِيلِ اللَّهِ":"الْإِمَامُ الْأَوْزَاعِيُّ -رحمه الله-":"المستدرك للحاكم"

"كَانَ إِذَا نَزَلَ قَامَ شَطْرَ اللَّيْلِ":"عَبْدُ اللَّهِ بْنُ عَبَّاسٍ -رضي الله عنهما-":"صحيح البخاري (2290)"

"مَا نُزِلَ الْقُرْآنُ أَنْ يُقْرَأَ وَيُقْفَ عَلَى حُرُوفِهِ، وَإِنَّمَا نُزِلَ أَنْ يُعْمَلَ بِهِ":"الْإِمَامُ الشَّافِعِيُّ -رحمه الله-":"الأم للشافعي"

"مَا جَالَسَ الْقُرْآنَ أَحَدٌ إِلَّا قَامَ مِنْ عِنْدِهِ بِزِيَادَةٍ أَوْ نُقْصَانٍ":"الْحَسَنُ الْبَصْرِيُّ -رحمه الله-":"حلية الأولياء (2/158)"

"اقْرَءُوا الْقُرْآنَ وَلَا تَغُرَّنَّكُمْ هَٰذِهِ الْمَصَاحِفُ الْمُعَلَّقَةُ، فَإِنَّ اللَّهَ لَا يُعَذِّبُ قَلْبًا هُوَ وَعَاءٌ لِلْقُرْآنِ":"أَبُو أُمَامَةَ الْبَاهِلِيُّ -رضي الله عنه-":"مصنف ابن أبي شيبة"

"إِنَّمَا هَذِهِ الْقُلُوبُ أَوْعِيَةٌ، فَاشْغَلُوهَا بِالْقُرْآنِ وَلَا تَشْغَلُوهَا بِغَيْرِهِ":"عَبْدُ اللَّهِ بْنُ مَسْعُودٍ -رضي الله عنه-":"مصنف ابن أبي شيبة (7/103)"

"إِنَّ الرَّجُلَ إِذَا طَلَبَ الْقُرْآنَ وَالْعِلْمَ لِلَّهِ لَمْ يَلْبَثْ أَنْ يُرَى ذَٰلِكَ فِي خُشُوعِهِ وَزُهْدِهِ وَحِلْمِهِ وَتَوَاضُعِهِ":"الْحَسَنُ الْبَصْرِيُّ -رحمه الله-":"حلية الأولياء (2/158)"

"لَقَدْ عَشِينَا دَهْرًا طَوِيلًا وَأَحَدُنَا يُؤْتَى الْإِيمَانَ قَبْلَ الْقُرْآنِ":"عَبْدُ اللَّهِ بْنُ عَمْرِو بْنِ الْعَاصِ -رضي الله عنهما-":"مصنف ابن أبي شيبة"

"كُنَّا نَتَعَلَّمُ الْإِيمَانَ كَمَا نَتَعَلَّمُ السُّورَةَ":"عُمَرُ بْنُ الْخَطَّابِ -رضي الله عنه-":"مصنف ابن أبي شيبة، المستدرك للحاكم"

"إِنَّ اللَّهَ لَيَرْضَى عَنِ الْعَبْدِ يَأْكُلُ الْأَكْلَةَ فَيَحْمَدُهُ عَلَيْهَا، أَوْ يَشْرَبُ الشَّرْبَةَ فَيَحْمَدُهُ عَلَيْهَا":"النَّبِيُّ ﷺ":"سُنَنُ التِّرْمِذِيّ (3458)، سُنَنُ ابْنِ مَاجَهْ (929)"

"أَكْثِرُوا مِنْ ذِكْرِ هَٰذِهِ النِّعَمِ، فَإِنَّ ذِكْرَهَا شُكْرٌ":"الْحَسَنُ الْبَصْرِيُّ -رحمه الله-":"الشكر لابن أبي الدنيا"

"مَا أُرِيدُ بِهِ وَجْهِيَ فَقَلِيلُهُ كَثِيرٌ، وَمَا أُرِيدُ بِهِ غَيْرِيَ فَكَثِيرُهُ قَلِيلٌ":"أَبُو هُرَيْرَةَ -رضي الله عنه-":"مصنف ابن أبي شيبة"

"الْعَبْدُ يَذْنَبُ الذَّنْبَ فَلَا يَزَالُ الشَّيْطَانُ يَقُولُ: اتَّقِ اللَّهَ...":"عَلِيُّ بْنُ أَبِي طَالِبٍ -رضي الله عنه-":"مصنف ابن أبي شيبة"

"رُبَّ عَمَلٍ صَغِيرٍ تُعْظِمُهُ النِّيَّةُ، وَرُبَّ عَمَلٍ كَبِيرٍ تُصَغِّرُهُ النِّيَّةُ":"عَبْدُ اللَّهِ بْنُ الْمُبَارَكِ -رحمه الله-":"حلية الأولياء (8/63)"

"إِنَّمَا يَخْشَى اللَّهَ مِنْ عِبَادِهِ الْعُلَمَاءُ":"الْفُضَيْلُ بْنُ عِيَاضٍ -رحمه الله-":"حلية الأولياء (7/33)"

"إِنَّ مِنْ شَرِّ النَّاسِ أَقْوَامًا قَرَءُوا الْقُرْآنَ لَا يَعْمَلُونَ بِسُنَّتِهِ...":"الْحَسَنُ الْبَصْرِيُّ -رحمه الله-":"حلية الأولياء (2/158)"

"إِنِّي لَأَمْقُتُ الرَّجُلَ أَنْ أَرَاهُ فَارِغًا لَيْسَ فِي شَيْءٍ مِنْ عَمَلِ الدُّنْيَا وَلَا عَمَلِ الْآخِرَةِ":"عَبْدُ اللَّهِ بْنُ مَسْعُودٍ -رضي الله عنه-":"مصنف ابن أبي شيبة"

"مَا عَالَجْتُ شَيْئًا أَشَدَّ عَلَيَّ مِنْ نِيَّتِي، لِأَنَّهَا تَتَقَلَّبُ عَلَيَّ":"سُفْيَانُ الثَّوْرِيُّ -رحمه الله-":"حلية الأولياء (6/364)"

"إِنَّمَا يَخْشَى اللَّهَ مِنْ عِبَادِهِ الْعُلَمَاءُ":"الْفُضَيْلُ بْنُ عِيَاضٍ -رحمه الله-":"حلية الأولياء (7/33)"

"اللَّهُمَّ إِنِّي أَعُوذُ بِكَ مِنَ الْعَجْزِ وَالْكَسَلِ":"النَّبِيُّ ﷺ":"صحيح البخاري (6363)، صحيح مسلم (2709)"

"كَانَ النَّبِيُّ ﷺ إِذَا أَنَامَ مِنَ اللَّيْلِ أَوْ مَرِضَ صَلَّى فِي النَّهَارِ ثِنْتَيْ عَشْرَةَ رَكْعَةً":"عَائِشَةُ -رضي الله عنها-":"صحيح مسلم (719)"

"إِنَّ الْإِيمَانَ نِصْفَانِ: نِصْفٌ صَبْرٌ، وَنِصْفٌ شُكْرٌ":"عَبْدُ اللَّهِ بْنُ مَسْعُودٍ -رضي الله عنه-":"مصنف ابن أبي شيبة"

"مَا يَأْتِي عَلَيَّ يَوْمٌ لَا أُرْمَى فِيهِ بِعَظِيمَةٍ إِلَّا عَدَدْتُهَا مِنَ اللَّهِ نِعْمَةً":"أَبُو الدَّرْدَاءِ -رضي الله عنه-":"مصنف ابن أبي شيبة"

"إِنَّ اللَّهَ لَيُمَتِّعُ بِالنِّعْمَةِ مَا شَاءَ، فَإِذَا لَمْ يُشْكَرْ قَلَبَهَا عَلَيْهِمْ عِقَابًا":"الْحَسَنُ الْبَصْرِيُّ -رحمه الله-":"حلية الأولياء (2/158)"

"فَإِنَّ الْإِنْسَانَ إِذَا قَرَأَ الْقُرْآنَ وَتَدَبَّرَهُ كَانَ ذَٰلِكَ مِنْ أَقْوَى الْأَسْبَابِ الْمَانِعَةِ لَهُ مِنَ الْمَعَاصِي":"شَيْخُ الْإِسْلَامِ ابْنُ تَيْمِيَّةَ -رحمه الله-":"الفتاوى الكبرى (5/532)"

"الشُّكْرُ مَبْنِيٌّ عَلَى خَمْسِ قَوَاعِدَ: خُضُوعِ الشَّاكِرِ، وَحُبِّهِ، وَاعْتِرَافِهِ بِنِعْمَتِهِ، وَالثَّنَاءِ عَلَيْهِ بِهَا، وَأَنْ لَا يَسْتَعْمِلَهَا فِيمَا يَكْرَهُ":"ابْنُ الْقَيِّمِ -رحمه الله-":"مدارج السالكين (2/244)"

"لَا يَنْبَغِي لِأَحَدٍ أَنْ يَقُولَ: أَنَا أَعْلَمُ، وَلَكِنْ يَقُولَ: نَحْنُ نَعْلَمُ أَوْ نَجْهَلُ":"الْإِمَامُ مَالِكُ بْنُ أَنَسٍ -رحمه الله-":"الجامع للبخاري، المدونة"

"مَا رَأَيْتُ شَيْئًا يُغَذِّي الْعَقْلَ وَالرُّوحَ، وَيَحْفَظُ الْجَسَدَ، وَيَضْمَنُ السَّعَادَةَ أَكْثَرَ مِنْ إِدَامَةِ النَّظَرِ فِي كِتَابِ اللَّهِ":"شَيْخُ الْإِسْلَامِ ابْنُ تَيْمِيَّةَ -رحمه الله-":"الفتاوى الكبرى"

"وَقَلَّ أَهْلُهُ فِي الْعَالَمِينَ":"ابْنُ الْقَيِّمِ -رحمه الله-":"مدارج السالكين (2/244)"

"كَانُوا يَتَعَلَّمُونَ النِّيَّةَ لِلْعَمَلِ كَمَا تَتَعَلَّمُونَ الْعَمَلَ":"سُفْيَانُ الثَّوْرِيُّ -رحمه الله-":"حلية الأولياء (6/364)"

"لَا يَزَالُ الْعَبْدُ بِخَيْرٍ مَا إِذَا قَالَ قَالَ لِلَّهِ، وَإِذَا عَمِلَ عَمِلَ لِلَّهِ":"الْحَسَنُ الْبَصْرِيُّ -رحمه الله-":"حلية الأولياء (2/158)"

"مَنْ أَحَبَّ اللَّهَ أَحَبَّ الْقُرْآنَ":"سُفْيَانُ بْنُ عُيَيْنَةَ -رحمه الله-":"حلية الأولياء (7/33)"

"لَيْسَ لِلْمُؤْمِنِ رَاحَةٌ دُونَ لِقَاءِ اللَّهِ":"عَبْدُ اللَّهِ بْنُ مَسْعُودٍ -رضي الله عنه-":"مصنف ابن أبي شيبة"

"إِنَّ الرَّجُلَ إِذَا طَلَبَ الْقُرْآنَ لِلَّهِ لَمْ يَلْبَثْ أَنْ يُرَى ذَٰلِكَ فِي خُشُوعِهِ":"الْحَسَنُ الْبَصْرِيُّ -رحمه الله-":"حلية الأولياء (2/158)"

"تَعَوَّدُوا الْخَيْرَ، فَإِنَّمَا الْخَيْرُ فِي الْعَادَةِ":"عَبْدُ اللَّهِ بْنُ مَسْعُودٍ -رضي الله عنه-":"مصنف ابن أبي شيبة (37291)"

"إِنَّ لِلَّهِ أَهْلِينَ مِنَ النَّاسِ... هُمْ أَهْلُ الْقُرْآنِ، أَهْلُ اللَّهِ وَخَاصَّتُهُ":"أَنَسُ بْنُ مَالِكٍ -رضي الله عنه-":"صحيح ابن ماجه"

"حَامِلُ الْقُرْآنِ حَامِلُ رَايَةِ الْإِسْلَامِ، لَا يَنْبَغِي أَنْ يَلْهُوَ مَعَ مَنْ يَلْهُوَ، وَلَا يَسْهُوَ مَعَ مَنْ يَسْهُوَ، وَلَا يَلْغُوَ مَعَ مَنْ يَلْغُوَ، تَعْظِيمًا لِحَقِّ الْقُرْآنِ":"الْفُضَيْلُ بْنُ عِيَاضٍ -رحمه الله-":"حلية الأولياء (7/33)"

"قُرَّاءُ الْقُرْآنِ ثَلَاثَةٌ: فَصِنْفٌ اتَّخَذُوهُ بِضَاعَةً يَأْكُلُونَ بِهَا، وَصِنْفٌ أَقَامُوا حُرُوفَهُ وَضَيَّعُوا حُدُودَهُ... وَصِنْفٌ عَمِدُوا إِلَى دَوَاءِ الْقُرْآنِ فَوَضَعُوهُ عَلَى دَاءِ قُلُوبِهِمْ... فَأُولَٰئِكَ الَّذِينَ يُسَقِي اللَّهُ بِهِمُ الْغَيْثَ، وَيَنْصُرُ بِهِمْ عَلَى الْأَعْدَاءِ":"الْحَسَنُ الْبَصْرِيُّ -رحمه الله-":"حلية الأولياء (2/158)"

"لَوْ طَهُرَتْ قُلُوبُنَا لَمْ تَشْبَعْ مِنْ قِرَاءَةِ كِتَابِ اللَّهِ":"عُثْمَانُ بْنُ عَفَّانَ -رضي الله عنه-":"حلية الأولياء (1/78)"

"أَفَلَا أَكُونُ عَبْدًا شَكُورًا؟":"النَّبِيُّ ﷺ":"صحيح البخاري (1130)، صحيح مسلم (2820)"

"مَا أَنْعَمَ اللَّهُ عَلَى عَبْدٍ نِعْمَةً فَقَالَ: الْحَمْدُ لِلَّهِ، إِلَّا كَانَ مَا أُعْطِيَ أَكْثَرَ مِمَّا أُخِذَ":"الْحَسَنُ الْبَصْرِيُّ -رحمه الله-":"الشكر"

"إِنْ أَصْبَحْتَ فَلَا تَنْتَظِرِ الْمَسَاءَ، وَإِنْ أَمْسَيْتَ فَلَا تَنْتَظِرِ الصَّبَاحَ":"عَبْدُ اللَّهِ بْنُ عُمَرَ -رضي الله عنهما-":"صحيح البخاري"

"رُبَّ عَمَلٍ صَغِيرٍ تُعْظِمُهُ النِّيَّةُ، وَرُبَّ عَمَلٍ كَبِيرٍ تُصَغِّرُهُ النِّيَّةُ":"عَبْدُ اللَّهِ بْنُ الْمُبَارَكِ -رحمه الله-":"حلية الأولياء (8/63)"

"وَاللَّهِ مَا دُونَ الْقُرْآنِ مِنْ غِنًى وَلَا بَعْدَهُ مِنْ فَاقَةٍ":"الْحَسَنُ الْبَصْرِيُّ -رحمه الله-":"حلية الأولياء (2/158)"

"نَحْنُ نَتَعَلَّمُ الْعِلْمَ وَالْحِلْمَ فِي آنٍ وَاحِدٍ":"الْإِمَامُ مَالِكُ بْنُ أَنَسٍ -رحمه الله-":"الجامع للبخاري"
